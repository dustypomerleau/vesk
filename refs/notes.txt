Because I am using an {#each} block to create the cards, I'm not really sure how to implement context. Would you leave this prop-drilled, or is there a better way?

<!-- LinkButton.svelte -->
<script>
    export let buttonText;
    export let href;
    export let icon;
</script>

<a {href} >
    <div>
        <svelte:component this={icon} />
    </div>
    <div>{buttonText}</div>
</a>

<!-- Card.svelte -->
<script>
    export let bulletPoints;
    export let buttonText;
    export let href;
    export let icon;
    export let title;
    import LinkButton from "$lib/LinkButton.svelte";
</script>

<div>
    <div>
        <h2>{title}</h2>
        <ul>
            {#each bulletPoints as point}
                <li>{@html point}</li>
            {/each}
        </ul>
    </div>
    <LinkButton {buttonText} {href} {icon} />
</div>

<!-- index.svelte -->
<script>
    import Card from "$lib/Card.svelte";
    import { cards } from "$lib/cardContent";
</script>

<div>
    {#each cards as card}
        <Card {...card} />
    {/each}
</div>
